cmake_minimum_required(VERSION 3.19)
project(event_flicker LANGUAGES CXX)
add_library(event_flicker src/filter.cpp src/pink.cpp)
add_library(event_flicker::event_flicker ALIAS event_flicker)
target_compile_features(event_flicker PUBLIC cxx_std_17)
target_include_directories(event_flicker PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
option(EVENT_FLICKER_BUILD_TESTS "Build standalone filter tests and benchmark" OFF)
if(EVENT_FLICKER_BUILD_TESTS)
    enable_testing()
    add_executable(event_flicker_tests tests/filter_test.cpp)
    target_link_libraries(event_flicker_tests PRIVATE event_flicker)
    add_test(NAME event_flicker_tests COMMAND event_flicker_tests)
    add_executable(event_flicker_benchmark tests/benchmark.cpp)
    target_link_libraries(event_flicker_benchmark PRIVATE event_flicker)
    add_executable(event_flicker_pink_tests tests/pink_filter_test.cpp)
    target_link_libraries(event_flicker_pink_tests PRIVATE event_flicker)
    add_test(NAME event_flicker_pink_tests COMMAND event_flicker_pink_tests)
endif()
option(EVENT_FLICKER_BUILD_EXAMPLE "Build text stream adapter example" OFF)
if(EVENT_FLICKER_BUILD_EXAMPLE)
    add_executable(event_flicker_stream examples/stream.cpp)
    target_link_libraries(event_flicker_stream PRIVATE event_flicker)
endif()
